export const overview = asyncHandler(async (_req, res) => {
  const [users, volunteers, reports, pendingReports, openReports, adoptions, shelters] = await Promise.all([
    User.countDocuments(),
    User.countDocuments({ role: 'volunteer' }),
    AnimalReport.countDocuments(),
    AnimalReport.countDocuments({ verified: false }),
    AnimalReport.countDocuments({ status: { $ne: 'Resolved' } }),
    Adoption.countDocuments({ adoptionStatus: 'Available' }),
    Shelter.countDocuments(),
  ]);
  const recentReports = await AnimalReport.find()
    .sort({ date: -1 })
    .populate('reporterId', 'name email phone')
    .populate('assignedVolunteerId', 'name');
  const people = await User.find().select('-password').sort({ createdAt: -1 });
  res.json({
    stats: { users, volunteers, reports, pendingReports, openReports, availableAdoptions: adoptions, shelters },
    recentReports,
    usersList: people,
  });
});

export const updateUser = asyncHandler(async (req, res) => {
  const user = await User.findById(req.params.id);
  if (!user) throw new HttpError(404, 'User not found.');
  if (user._id.equals(req.user._id) && req.body.isActive === false)
    throw new HttpError(400, 'You cannot deactivate your own admin account.');
  if (req.body.role && ['user', 'volunteer', 'shelter', 'admin'].includes(req.body.role))
    user.role = req.body.role;
  if (typeof req.body.isActive === 'boolean') user.isActive = req.body.isActive;
  if (req.body.phone) user.phone = req.body.phone;
  await user.save();
  if (user.role === 'volunteer')
    await Volunteer.findOneAndUpdate(
      { userId: user._id },
      {
        $set: {
          userId: user._id,
          name: user.name,
          contact: user.phone,
          area: 'Set your service area',
        },
      },
      { upsert: true }
    );
  res.json({
    user: {
      id: user._id,
      name: user.name,
      email: user.email,
      role: user.role,
      phone: user.phone,
      isActive: user.isActive,
    },
  });
});

export const deleteUser = asyncHandler(async (req, res) => {
  if (String(req.user._id) === req.params.id)
    throw new HttpError(400, 'You cannot delete your own account.');
  const user = await User.findByIdAndDelete(req.params.id);
  if (!user) throw new HttpError(404, 'User not found.');
  await Volunteer.deleteOne({ userId: user._id });
  res.json({ message: 'User account removed.' });
});

export const listAdoptions = asyncHandler(async (req, res) => {

    if (!req.user.shelterId) return res.json({ adoptions: [] });
    query.shelterId = req.user.shelterId;
    if (req.query.status !== 'all') query.adoptionStatus = req.query.status || 'Available';
  } else if (req.user?.role === 'admin') {
    if (req.query.status && req.query.status !== 'all') query.adoptionStatus = req.query.status;
  } else {
    query.adoptionStatus = 'Available';
  }
  const adoptions = await Adoption.find(query)
    .sort({ createdAt: -1 })
    .populate('shelterId', 'name address phone email');
  res.json({ adoptions });
});

export const createAdoption = asyncHandler(async (req, res) => {
  const shelterId = req.user.role === 'admin' ? req.body.shelterId : req.user.shelterId;
  if (req.user.role !== 'admin') {
    if (!shelterId)
      throw new HttpError(400, 'Add your shelter details before creating an adoption listing.');
    const shelter = await Shelter.findById(shelterId);
    if (!shelter)
      throw new HttpError(400, 'Add your shelter details before creating an adoption listing.');
  }
  const { animalType, age, gender, description } = req.body;
  if (!animalType?.trim() || !age?.trim() || !description?.trim() || !shelterId)
    throw new HttpError(400, 'Animal type, age, description and shelter are required.');
  const adoption = await Adoption.create({
    animalType,
    age,
    gender,
    description,
    shelterId,
    photo: req.file ? `/uploads/${req.file.filename}` : '',
  });
  res.status(201).json({ adoption });
});

export const updateAdoption = asyncHandler(async (req, res) => {
  const adoption = await Adoption.findById(req.params.id);
  if (!adoption) throw new HttpError(404, 'Adoption listing not found.');
  if (req.user.role !== 'admin' && String(adoption.shelterId) !== String(req.user.shelterId))
    throw new HttpError(403, 'You can only manage listings from your shelter.');
  const allowed = ['animalType', 'age', 'gender', 'description', 'adoptionStatus'];
  for (const key of allowed) if (req.body[key] !== undefined) adoption[key] = req.body[key];
  if (req.file) adoption.photo = `/uploads/${req.file.filename}`;
  await adoption.save();
  res.json({ adoption });
});

export const register = asyncHandler(async (req, res) => {
  const { name, email, phone, password } = req.body;
  if (!name?.trim() || !email?.trim() || !phone?.trim() || !password) {
    throw new HttpError(400, 'Name, email, phone, and password are required.');
  }
  if (password.length < 8) throw new HttpError(400, 'Password must be at least 8 characters.');

  const SIGNUP_ROLES = ['user', 'volunteer', 'shelter'];
  const role = SIGNUP_ROLES.includes(req.body.role) ? req.body.role : 'user';

  const normalizedEmail = email.trim().toLowerCase();
  if (await User.exists({ email: normalizedEmail }))
    throw new HttpError(409, 'An account with this email already exists.');
  const passwordHash = await bcrypt.hash(password, 12);
  const user = await User.create({
    name: name.trim(),
    email: normalizedEmail,
    phone: phone.trim(),
    password: passwordHash,
    role,
  });
  res.status(201).json({ token: signToken(user), user: publicUser(user) });
});

export const login = asyncHandler(async (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) throw new HttpError(400, 'Email and password are required.');
  const user = await User.findOne({ email: email.trim().toLowerCase() }).select('+password');
  if (!user || !(await bcrypt.compare(password, user.password)))
    throw new HttpError(401, 'Email or password is incorrect.');
  if (!user.isActive)
    throw new HttpError(403, 'This account has been paused. Please contact PawCare support.');
  res.json({ token: signToken(user), user: publicUser(user) });
});

const RESET_MINUTES = 30;
const RESET_COOLDOWN_MS = 60 * 1000;
const GENERIC_RESET_MESSAGE =
  'If an account exists for that email, a password reset link is on its way.';

const hashToken = (token) => crypto.createHash('sha256').update(token).digest('hex');

export const forgotPassword = asyncHandler(async (req, res) => {
  const email = req.body.email?.trim().toLowerCase();
  if (!email) throw new HttpError(400, 'Please enter your email address.');

  const user = await User.findOne({ email }).select('+resetLastSentAt');
  const cooledDown =
    !user?.resetLastSentAt || Date.now() - user.resetLastSentAt.getTime() > RESET_COOLDOWN_MS;

  if (user && cooledDown) {
    const token = crypto.randomBytes(32).toString('hex');
    await User.updateOne(
      { _id: user._id },
      {
        $set: {
          resetTokenHash: hashToken(token),
          resetExpires: new Date(Date.now() + RESET_MINUTES * 60 * 1000),
          resetLastSentAt: new Date(),
        },
      }
    );
    const base = process.env.FRONTEND_URL || 'http://localhost:3000';
    try {
      await sendPasswordResetEmail(user.email, `${base}/reset-password?token=${token}`);
    } catch (error) {
      console.error('Password reset email failed:', error.message);
    }
  }

  // Same response whether or not the account exists, so the form can't reveal who has an account.
  res.json({ message: GENERIC_RESET_MESSAGE });
});

export const resetPassword = asyncHandler(async (req, res) => {
  const { token, password } = req.body;
  if (!token || !password) throw new HttpError(400, 'Reset link and new password are required.');
  if (password.length < 8) throw new HttpError(400, 'Password must be at least 8 characters.');

  const user = await User.findOne({
    resetTokenHash: hashToken(token),
    resetExpires: { $gt: new Date() },
  });
  if (!user)
    throw new HttpError(400, 'This reset link is invalid or has expired. Please request a new one.');

  const passwordHash = await bcrypt.hash(password, 12);
  await User.updateOne(
    { _id: user._id },
    {
      $set: { password: passwordHash },
      $unset: { resetTokenHash: '', resetExpires: '', resetLastSentAt: '' },
    }
  );

  res.json({ message: 'Your password has been reset. You can sign in now.' });
});

const rescueOrder = (a, b) =>

export const listReports = asyncHandler(async (req, res) => {

  if (req.query.status) query.status = req.query.status;
  if (req.query.priority) query.priority = req.query.priority;
  const reports = await AnimalReport.find(query)
    .sort({ priority: -1, date: 1 })
    .populate('reporterId', 'name')
    .populate('assignedVolunteerId', 'name');
  // Explicit domain ranking avoids relying on alphabetic enum order in MongoDB.
  reports.sort(rescueOrder);
  const isAssignedVolunteer = req.user?.role === 'volunteer';
  res.json({
    reports: reports.map((report) => {
      const item = report.toObject();
      const maySeeContact =
        req.user?.role === 'admin' ||
        (isAssignedVolunteer && String(report.assignedVolunteerId?._id) === String(req.user._id));

      return item;
    }),
  });
});

export const updateStatus = asyncHandler(async (req, res) => {
  const allowed = ['Accepted', 'In Progress', 'Resolved'];
  if (!allowed.includes(req.body.status))
    throw new HttpError(400, 'Choose Accepted, In Progress, or Resolved.');
  const report = await AnimalReport.findById(req.params.id);
  if (!report) throw new HttpError(404, 'Rescue report not found.');
  const isManager = req.user.role === 'admin' || req.user.role === 'shelter';
  if (!isManager && String(report.assignedVolunteerId) !== String(req.user._id))
    throw new HttpError(403, 'Only the assigned volunteer can update this rescue.');
  const transitions = {
    Pending: ['Accepted'],
    Accepted: ['In Progress'],
    'In Progress': ['Resolved'],
    Resolved: [],
  };
  if (!isManager && !transitions[report.status].includes(req.body.status))
    throw new HttpError(409, `A case marked ${report.status} cannot move to ${req.body.status}.`);
  report.status = req.body.status;
  await report.save();
  res.json({ message: `Rescue status updated to ${report.status}.`, report });
});

import crypto from 'node:crypto';
import { User } from '../models/User.js';
import { HttpError, asyncHandler } from '../http.js';
import { sendVerificationEmail } from '../mailer.js';

const CODE_MINUTES = 10;
const MAX_ATTEMPTS = 5;

function hashCode(code) {
  return crypto.createHash('sha256').update(code).digest('hex');
}

export const sendVerificationCode = asyncHandler(async (req, res) => {
  const user = await User.findById(req.user._id).select('+verifyCodeHash');
  if (!user) throw new HttpError(404, 'Account not found.');
  if (user.emailVerified) return res.json({ message: 'Your email is already verified.' });

  const code = String(crypto.randomInt(0, 1000000)).padStart(6, '0');

  // updateOne avoids re-validating the whole user, including the password field.
  await User.updateOne(
    { _id: user._id },
    {
      $set: {
        verifyCodeHash: hashCode(code),
        verifyExpires: new Date(Date.now() + CODE_MINUTES * 60_000),
        verifyAttempts: 0,
      },
    }
  );

  await sendVerificationEmail(user.email, code);
  res.json({ message: 'We sent a 6-digit code to your email.' });
});

export const verifyEmailCode = asyncHandler(async (req, res) => {
  const code = String(req.body.code || '').trim();
  if (!/^\d{6}$/.test(code)) throw new HttpError(400, 'Enter the 6-digit code from your email.');

  const user = await User.findById(req.user._id).select('+verifyCodeHash +verifyExpires +verifyAttempts');
  if (!user) throw new HttpError(404, 'Account not found.');
  if (user.emailVerified) return res.json({ message: 'Your email is already verified.' });

  if (!user.verifyCodeHash || !user.verifyExpires || user.verifyExpires < new Date())
    throw new HttpError(400, 'This code has expired. Request a new one.');

  if (user.verifyAttempts >= MAX_ATTEMPTS)
    throw new HttpError(429, 'Too many attempts. Request a new code.');

  if (hashCode(code) !== user.verifyCodeHash) {
    await User.updateOne({ _id: user._id }, { $inc: { verifyAttempts: 1 } });
    throw new HttpError(400, 'That code is incorrect.');
  }

  await User.updateOne(
    { _id: user._id },
    {
      $set: { emailVerified: true },
      $unset: { verifyCodeHash: '', verifyExpires: '', verifyAttempts: '' },
    }
  );
  res.json({ message: 'Your email is verified.' });
});

import { AnimalReport } from '../models/AnimalReport.js';

export const volunteerSummary = asyncHandler(async (req, res) => {
  let profile = await Volunteer.findOne({ userId: req.user._id });
  if (!profile && req.user.role === 'volunteer') {

import fs from 'node:fs';
import { checkAnimalImage } from '../imageCheck.js';

// Used on routes where a photo is required (adoption listings).
// Rejects bad photos, otherwise stores the result on req.imageCheck.
export async function checkAnimalPhoto(req, res, next) {
  if (!req.file) {
    return res.status(400).json({ message: 'Please attach a photo of the animal.' });
  }

  try {
    const check = await checkAnimalImage(req.file.path);

    if (check.status === 'rejected') {
      await fs.promises.unlink(req.file.path).catch(() => {});
      return res.status(422).json({ message: check.reason });
    }

    req.imageCheck = check;
    next();
  } catch (error) {
    await fs.promises.unlink(req.file.path).catch(() => {});
    next(error);
  }
}

// Used on routes where a photo is optional (lost and found reports).
// Skips the check when no photo was sent.
export async function checkOptionalAnimalPhoto(req, res, next) {
  if (!req.file) return next();
  return checkAnimalPhoto(req, res, next);
}

export function signToken(user) {
  return jwt.sign({ id: user._id.toString(), role: user.role }, jwtSecret(), {
    expiresIn: process.env.JWT_EXPIRES_IN || '7d',
  });

async function attachUser(req, optional = false) {
  const token = req.headers.authorization?.startsWith('Bearer ')
    ? req.headers.authorization.slice(7)
    : null;
  if (!token) {
    if (optional) return null;
    throw new HttpError(401, 'Please sign in to continue.');
  }
  let payload;
  try {
    payload = jwt.verify(token, jwtSecret());
  } catch {
    throw new HttpError(401, 'Your session is invalid or has expired. Please sign in again.');
  }
  const user = await User.findById(payload.id).select('-password');
  if (!user || !user.isActive) throw new HttpError(401, 'This account is unavailable.');
  req.user = user;
  return user;
}

/** Require a signed bearer token and resolve its current database user. */
export const authenticate = asyncHandler(async (req, _res, next) => {
  await attachUser(req);
  next();
});

  'mailinator.com',

export function checkEmailDomain(email) {

  if (TYPO_FIXES[domain])

===== \backend\imageCheck.js =====
import fs from 'node:fs';
import crypto from 'node:crypto';
import sharp from 'sharp';
import { pipeline, RawImage } from '@huggingface/transformers';

const MODEL_ID = 'Xenova/mobilenet_v2_1.0_224';

// ImageNet classes 0 to 397 are animals. Classes 398 and up are objects.
const ANIMAL_MAX_INDEX = 397;

// Tune these after testing with real PawCare uploads.
const BLUR_THRESHOLD = Number(process.env.BLUR_THRESHOLD || 60);
const APPROVE_CONFIDENCE = Number(process.env.APPROVE_CONFIDENCE || 0.7);

let classifierPromise = null;
let animalLabels = new Set();

// Load the model once and reuse it. The first call downloads it.
function getClassifier() {
  if (!classifierPromise) {
    classifierPromise = pipeline('image-classification', MODEL_ID).then((classifier) => {
      const id2label = classifier.model.config.id2label;
      animalLabels = new Set(
        Object.entries(id2label)
          .filter(([index]) => Number(index) <= ANIMAL_MAX_INDEX)
          .map(([, label]) => label)
      );
      return classifier;
    });
  }
  return classifierPromise;
}

// Blur score: variance of a Laplacian-filtered grayscale image.
// Low variance means few sharp edges, which usually means a blurry photo.
async function blurScore(imageBuffer) {
  const { channels } = await sharp(imageBuffer)
    .grayscale()
    .convolve({ width: 3, height: 3, kernel: [0, 1, 0, 1, -4, 1, 0, 1, 0] })
    .stats();
  return channels[0].stdev ** 2;
}

/**
 * Checks an uploaded image file.
 * Returns { status, reason, suggestedSpecies?, confidence?, fileHash }
 * status is one of: 'approved', 'pending_review', 'rejected'
 */
export async function checkAnimalImage(filePath) {
  // Normalize size and rotation so the scores are consistent.
  const jpeg = await sharp(filePath)
    .rotate()
    .resize({ width: 1024, withoutEnlargement: true })
    .jpeg({ quality: 90 })
    .toBuffer();

  // Hash the original file, used later to detect repeat uploads.
  const fileHash = crypto
    .createHash('sha256')
    .update(await fs.promises.readFile(filePath))
    .digest('hex');

  // 1. Blur check
  const blur = await blurScore(jpeg);
  if (blur < BLUR_THRESHOLD) {
    return {
      status: 'rejected',
      reason: 'The photo is too blurry. Please upload a clearer picture.',
      fileHash,
    };
  }

  // 2. Animal check
  const classifier = await getClassifier();
  const image = await RawImage.fromBlob(new Blob([jpeg], { type: 'image/jpeg' }));
  const results = await classifier(image, { top_k: 10 });

  const topAnimal = results.find((result) => animalLabels.has(result.label));

  if (!topAnimal) {
    return {
      status: 'rejected',
      reason: 'No animal was detected in this photo. Please upload a photo of the animal.',
      fileHash,
    };
  }

  // 3. Decide based on confidence
  const confident = topAnimal.score >= APPROVE_CONFIDENCE;

  return {
    status: confident ? 'approved' : 'pending_review',
    reason: confident ? 'Animal detected.' : 'Needs a quick review by our team.',
    suggestedSpecies: topAnimal.label,
    confidence: Number(topAnimal.score.toFixed(2)),
    fileHash,
  };
}

import crypto from 'node:crypto';

export const CODE_TTL_MS = 10 * 60_000;
export const COOLDOWN_MS = 60_000;
export const MAX_ATTEMPTS = 5;

function secret() {
  const value = process.env.PAWCARE_JWT_SECRET;
  if (!value && process.env.NODE_ENV === 'production')
    throw new Error('PAWCARE_JWT_SECRET must be set in production.');
  return value || 'local-dev-only-secret';
}

export function newCode() {
  return String(crypto.randomInt(0, 1_000_000)).padStart(6, '0');
}

// Keyed hash: without the server secret, a stolen database can't be used to recover codes.
export function hashCode(code) {
  return crypto.createHmac('sha256', secret()).update(code).digest('hex');
}

export function codeMatches(code, storedHash) {
  const a = Buffer.from(hashCode(code), 'hex');
  const b = Buffer.from(storedHash, 'hex');
  return a.length === b.length && crypto.timingSafeEqual(a, b);
}

// Mapbox uses [lng, lat], the reverse of Leaflet's [lat, lng].
const defaultCenter = [78.9629, 20.5937];
const noReports = [];

function createPawElement(selected = false) {
  const el = document.createElement('div');
  el.className = selected ? 'paw-map-icon selected' : 'paw-map-icon';
  el.innerHTML = '<span></span>';
  el.style.width = '24px';
  el.style.height = '28px';
  return el;
}

/** Interactive pin-selection map; when read-only, report markers open their short case summary. */
export default function AnimalMap({
  location,
  onSelect,
  reports = noReports,
  zoom,
  markerLabel = 'Selected animal location',
}) {
  const containerRef = useRef(null);
  const mapRef = useRef(null);
  const selectedMarkerRef = useRef(null);
  const reportMarkersRef = useRef([]);
  const onSelectRef = useRef(onSelect);

  const hasLocation = location?.lat != null && location?.lng != null;
  const lat = hasLocation ? Number(location.lat) : null;
  const lng = hasLocation ? Number(location.lng) : null;
  const mapZoom = zoom || (onSelect ? (hasLocation ? 13 : 4) : 5);

  // Keep the latest callback without recreating the map.
  useEffect(() => {
    onSelectRef.current = onSelect;
  }, [onSelect]);

  // Create the map once.
  useEffect(() => {
    const map = new mapboxgl.Map({
      container: containerRef.current,
      style: 'mapbox://styles/mapbox/streets-v12',
      center: hasLocation ? [lng, lat] : defaultCenter,
      zoom: mapZoom,
    });

    map.addControl(new mapboxgl.NavigationControl({ showCompass: false }), 'top-right');
    map.scrollZoom.disable();

    map.on('click', (event) => {
      const clickLat = event.lngLat.lat;
      const clickLng = event.lngLat.lng;
      onSelectRef.current?.({
        lat: Number(clickLat.toFixed(5)),
        lng: Number(clickLng.toFixed(5)),
        address: `Map pin Â· ${clickLat.toFixed(3)}, ${clickLng.toFixed(3)}`,
      });
    });

    mapRef.current = map;
    return () => {
      map.remove();
      mapRef.current = null;
    };
    // The map is created once; later changes are handled by the effects below.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // Re-center when the selected location changes (the Leaflet version remounted for this).
  useEffect(() => {
    const map = mapRef.current;
    if (!map || !hasLocation) return;
    map.easeTo({ center: [lng, lat], zoom: mapZoom });
  }, [lat, lng]);

  // Show the selected pin.
  useEffect(() => {
    const map = mapRef.current;
    if (!map) return;

    selectedMarkerRef.current?.remove();
    selectedMarkerRef.current = null;

    if (hasLocation) {
      selectedMarkerRef.current = new mapboxgl.Marker({
        element: createPawElement(true),
        anchor: 'bottom',
      })
        .setLngLat([lng, lat])
        .setPopup(new mapboxgl.Popup({ offset: 22 }).setText(markerLabel))
        .addTo(map);
    }
  }, [lat, lng, markerLabel]);

  // Show a paw marker for each report that has a location.
  useEffect(() => {
    const map = mapRef.current;
    if (!map) return;

    reportMarkersRef.current.forEach((marker) => marker.remove());
    reportMarkersRef.current = reports
      .filter((report) => report.location?.lat != null && report.location?.lng != null)
      .map((report) => {
        // Build the popup with textContent, not innerHTML, so report text can't inject HTML.
        const popupBody = document.createElement('div');
        popupBody.className = 'map-popup';

        const title = document.createElement('strong');
        title.textContent = `${report.animalType} Â· ${report.priority}`;

        const address = document.createElement('small');
        address.textContent = report.location.address || 'Location pin';

        popupBody.append(title, address);

        return new mapboxgl.Marker({ element: createPawElement(false), anchor: 'bottom' })
          .setLngLat([Number(report.location.lng), Number(report.location.lat)])
          .setPopup(new mapboxgl.Popup({ offset: 22 }).setDOMContent(popupBody))
          .addTo(map);
      });
  }, [reports]);

  return (
    <div className="map-frame" aria-label={`Mapbox ${markerLabel.toLowerCase()} map`}>
      {onSelect && <span className="map-help">Click the map to drop a location pin</span>}
      {onSelect && <LocationSearch onPick={onSelect} />}
      <div ref={containerRef} className="map-canvas" />
    </div>
  );
}

import { createContext, useContext, useEffect, useMemo, useState } from 'react';
import { api } from './client.js';

const AuthContext = createContext(null);
const TOKEN_KEY = 'pawcare-token';

export function AuthProvider({ children }) {
  const [token, setToken] = useState(() => localStorage.getItem(TOKEN_KEY) || '');
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(Boolean(token));

  useEffect(() => {
    let active = true;
    if (!token) {
      setUser(null);
      setLoading(false);
      return undefined;
    }
    api
      .me(token)
      .then(({ user: profile }) => {
        if (active) setUser(profile);
      })
      .catch(() => {
        if (active) {
          localStorage.removeItem(TOKEN_KEY);
          setToken('');
          setUser(null);
        }
      })
      .finally(() => {
        if (active) setLoading(false);
      });
    return () => {
      active = false;
    };
  }, [token]);

  // Stores the session. Only called once the email is verified.
  function startSession({ user: nextUser, token: nextToken }) {
    localStorage.setItem(TOKEN_KEY, nextToken);
    setUser(nextUser);
    setToken(nextToken);
  }

  // Unverified accounts get a token but no session, so the code screen can show first.
  async function login(payload) {
    const result = await api.login(payload);
    if (result.user.emailVerified) startSession(result);
    return result;
  }

  // Creates the account without signing in.
  async function register(payload) {
    return api.register(payload);
  }

  function logout() {
    localStorage.removeItem(TOKEN_KEY);
    setToken('');
    setUser(null);
  }

  const value = useMemo(
    () => ({ token, user, loading, login, register, logout, setUser, startSession }),
    [token, user, loading]
  );
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used inside AuthProvider.');
  return context;
}

// Code screen used after registering and after logging in to an unverified account.
function VerifyStep({ account, token, onDone }) {
  const { startSession } = useAuth();
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [busy, setBusy] = useState(false);
  const [cooldown, setCooldown] = useState(0);

  useEffect(() => {
    if (cooldown <= 0) return undefined;
    const timer = setTimeout(() => setCooldown((seconds) => seconds - 1), 1000);
    return () => clearTimeout(timer);
  }, [cooldown]);

  async function sendCode() {
    setError('');
    setNotice('');
    try {
      const result = await api.sendVerificationCode(token);
      setNotice(result.message);
      setCooldown(60);
    } catch (e) {
      setError(e.message);
    }
  }

  // Send the first code as soon as the screen opens.
  useEffect(() => {
    sendCode();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  async function submitCode(event) {
    event.preventDefault();
    setError('');
    setBusy(true);
    const code = String(new FormData(event.currentTarget).get('code') || '').trim();
    try {
      await api.verifyEmailCode(token, code);
      startSession({ user: { ...account, emailVerified: true }, token });
      onDone();
    } catch (e) {
      setError(e.message);
    } finally {
      setBusy(false);
    }
  }

  function verifyLater() {
    startSession({ user: account, token });
    onDone();
  }

  return (
    <main className="auth-section">

const API_ROOT = '/api';

/** Send a REST request and normalize JSON/error responses. */
export async function request(path, options = {}) {
  const { token, body, headers = {}, ...rest } = options;
  const nextHeaders = { ...headers };
  if (token) nextHeaders.Authorization = `Bearer ${token}`;
  const isForm = typeof FormData !== 'undefined' && body instanceof FormData;
  if (body && !isForm && !nextHeaders['Content-Type']) nextHeaders['Content-Type'] = 'application/json';
  let response;
  try {
    response = await fetch(`${API_ROOT}${path}`, {
      ...rest, method: rest.method || (body ? 'POST' : 'GET'), headers: nextHeaders,
      body: body ? (isForm ? body : JSON.stringify(body)) : undefined
    });
  } catch {
    throw new Error('PawCare could not reach the API. Check that the server is running and MongoDB is configured.');
  }
  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(data.message || `Request failed (${response.status}).`);
  return data;
}

import { useEffect, useRef, useState } from 'react';

const TOKEN = import.meta.env.VITE_MAPBOX_TOKEN;

export default function LocationSearch({ onPick }) {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);
  const [open, setOpen] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const requestIdRef = useRef(0);
  const chosenRef = useRef('');

  useEffect(() => {
    const text = query.trim();

    // Don't search again for the text we just picked from the list.
    if (text === chosenRef.current) return;

    if (text.length < 3) {
      setResults([]);
      setOpen(false);
      setError('');
      return;
    }

    const controller = new AbortController();
    // Wait a moment after typing stops before calling Mapbox, to save requests.
    const timer = setTimeout(async () => {
      const id = ++requestIdRef.current;
      setLoading(true);
      setError('');
      try {
        const url = new URL(
          `https://api.mapbox.com/geocoding/v5/mapbox.places/${encodeURIComponent(text)}.json`
        );
        url.searchParams.set('access_token', TOKEN);
        url.searchParams.set('autocomplete', 'true');
        url.searchParams.set('limit', '5');
        // Optional: uncomment and set your country code to limit results, e.g. 'in' for India.
        // url.searchParams.set('country', 'in');

        const res = await fetch(url, { signal: controller.signal });
        if (!res.ok) throw new Error('Search failed');
        const data = await res.json();

        // Ignore responses for older searches that finished late.
        if (id !== requestIdRef.current) return;
        setResults(data.features || []);
        setOpen(true);
      } catch (err) {
        if (err.name !== 'AbortError') {
          setError('Search is unavailable right now. You can still click the map.');
          setOpen(false);
        }
      } finally {
        setLoading(false);
      }
    }, 350);

    return () => {
      clearTimeout(timer);
      controller.abort();
    };
  }, [query]);

  function choose(feature) {
    const [lng, lat] = feature.center; // Mapbox returns [lng, lat]
    chosenRef.current = feature.place_name;
    setQuery(feature.place_name);
    setOpen(false);
    onPick({
      lat: Number(lat.toFixed(5)),
      lng: Number(lng.toFixed(5)),
      address: feature.place_name,
    });
  }

  return (
    <div className="map-search">
      <input
        type="search"
        value={query}
        onChange={(e) => {
          chosenRef.current = '';
          setQuery(e.target.value);
        }}
        onFocus={() => results.length && setOpen(true)}
        placeholder="Search a place or address"
        aria-label="Search for a location"
        autoComplete="off"
      />
      {loading && <small className="map-search-status">Searching...</small>}
      {error && <small className="map-search-status">{error}</small>}
      {open && results.length > 0 && (
        <ul className="map-search-results">
          {results.map((feature) => (
            <li key={feature.id}>
              <button type="button" onClick={() => choose(feature)}>
                {feature.place_name}
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

import { Navigate, Outlet, useLocation } from 'react-router-dom';
import { useAuth } from './AuthContext.jsx';

export default function ProtectedRoute({ roles }) {
  const { user, loading } = useAuth();
  const location = useLocation();
  if (loading) return <div className="page-loading" role="status">Checking your sign-inâ€¦</div>;
  if (!user) return <Navigate to="/login" replace state={{ from: location.pathname }} />;
  if (roles && !roles.includes(user.role)) return <Navigate to="/dashboard" replace />;
  return <Outlet />;
}

import { useState } from 'react';
import { api } from './client.js';
import { useAuth } from './AuthContext.jsx';
import AnimalMap from './AnimalMap.jsx';
import { Alert } from './UI.jsx';

export default function ReportForm({ rescueMode = false }) {
  const { token } = useAuth();
  const [location, setLocation] = useState({ lat: null, lng: null, address: '' });
  const [message, setMessage] = useState('');
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);

  async function handleSubmit(event) {
    event.preventDefault();
    setError('');
    setMessage('');
    const formElement = event.currentTarget;
    const form = new FormData(formElement);
    const pin = {
      ...location,
      address: form.get('address')?.trim() || location.address || 'Map pin location',
    };
    if (!Number.isFinite(pin.lat) || !Number.isFinite(pin.lng)) {
      setError("Choose the animal's location on the map before sending the report.");
      return;
    }
    form.set('location', JSON.stringify(pin));
    setBusy(true);
    try {
      const result = await api.createReport(token, form);
      setMessage(result.message || 'Your report has been sent for verification.');
      formElement.reset();
      setLocation({ lat: null, lng: null, address: '' });
    } catch (requestError) {
      setError(requestError.message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <form className="surface" onSubmit={handleSubmit}>
      <div className="surface-title">
        {rescueMode ? 'Tell us whatâ€™s happening' : 'Animal details'}
      </div>
      <p className="surface-copy">
        A clear description and location help nearby volunteers respond safely.
      </p>
      {error && (
        <div style={{ marginTop: 15 }}>
          <Alert kind="error">{error}</Alert>
        </div>
      )}
      {message && (
        <div style={{ marginTop: 15 }}>
          <Alert kind="success">{message} Track updates from your dashboard.</Alert>
        </div>
      )}
      <div className="form-grid" style={{ marginTop: 18 }}>
        <div className="form-field">
          <label htmlFor="animalType">Animal type *</label>
          <select id="animalType" name="animalType" required defaultValue="">
            <option value="" disabled>
              Select an animal
            </option>
            <option>Dog</option>
            <option>Cat</option>
            <option>Cow</option>
            <option>Bird</option>
            <option>Other</option>
          </select>
        </div>
        <div className="form-field">
          <label htmlFor="priority">Assistance priority *</label>
          <select
            id="priority"
            name="priority"
            required
            defaultValue={rescueMode ? 'Urgent' : 'Normal'}
          >
            <option value="Emergency">Emergency â€” severely injured</option>
            <option value="Urgent">Urgent â€” sick or injured</option>
            <option value="Normal">Normal â€” food or shelter</option>
          </select>
        </div>
        <div className="form-field full">
          <label htmlFor="description">What do you see? *</label>
          <textarea
            id="description"
            name="description"
            required
            minLength="10"
            maxLength="1800"
            placeholder="Describe the animal, condition, and anything that could help a volunteer prepare."
          />
        </div>
        <div className="form-field full">
          <label htmlFor="assistanceRequired">What help is needed? *</label>

        <div className="form-field full">
          <label htmlFor="address">Location / nearby address *</label>
          <input
            id="address"

import { useCallback, useEffect, useState } from 'react';
import { api } from './client.js';
import { useAuth } from './AuthContext.jsx';
import AnimalMap from './AnimalMap.jsx';
import DashboardLayout from './DashboardLayout.jsx';
import { Alert, EmptyState, Loading, PageSection, PriorityBadge, StatusBadge } from './UI.jsx';

export default function ShelterDashboard() {
  const { token, user, setUser } = useAuth();
  const [shelter, setShelter] = useState(null);
  const [requests, setRequests] = useState([]);
  const [listings, setListings] = useState([]);
  const [shelterLocation, setShelterLocation] = useState({ lat: null, lng: null, address: '' });
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [busy, setBusy] = useState(false);

  const refresh = useCallback(async () => {
    setError('');
    try {
      const [shelterData, requestData, adoptionData] = await Promise.all([
        api.shelters(),
        api.shelterRequests(token),
        api.myAdoptions(token),
      ]);
      const own =
        (shelterData.shelters || []).find((item) => String(item._id) === String(user.shelterId)) ||
        null;
      setShelter(own);
      setShelterLocation(own?.location || { lat: null, lng: null, address: '' });
      setRequests(requestData.reports || []);
      setListings(
        (adoptionData.adoptions || []).filter(
          (item) => String(item.shelterId?._id) === String(user.shelterId)
        )
      );
    } catch (e) {
      setError(e.message);
    } finally {
      setLoading(false);
    }
  }, [token, user]);

  useEffect(() => {
    refresh();
  }, [refresh]);

  async function saveShelter(event) {
    event.preventDefault();
    setBusy(true);
    setError('');
    setNotice('');
    const form = Object.fromEntries(new FormData(event.currentTarget));
    if (!Number.isFinite(shelterLocation.lat) || !Number.isFinite(shelterLocation.lng)) {
      setError('Choose the shelter location on the map before saving.');
      setBusy(false);
      return;
    }
    try {
      const result = shelter
        ? await api.updateShelter(token, shelter._id, form)
        : await api.createShelter(token, form);
      setShelter(result.shelter);
      setUser({ ...user, shelterId: result.shelter._id });
      setNotice('Shelter information saved.');
    } catch (e) {
      setError(e.message);
    } finally {
      setBusy(false);
    }
  }

  async function addListing(event) {
    event.preventDefault();
    setBusy(true);
    setError('');
    setNotice('');
    const formElement = event.currentTarget;
    const form = new FormData(formElement);
    try {
      const result = await api.createAdoption(token, form);
      setListings((current) => [...current, result.adoption]);
      formElement.reset();
      setNotice(
        result.adoption?.imageStatus === 'pending_review'
          ? 'Listing saved. The photo needs a quick review before it appears publicly.'
          : 'Adoption listing added.'
      );
    } catch (e) {
      setError(e.message);
    } finally {
      setBusy(false);
    }
  }

  async function changeStatus(item, status) {
    try {
      const result = await api.updateAdoption(token, item._id, { adoptionStatus: status });
      setListings((items) =>
        items.map((entry) => (entry._id === item._id ? result.adoption : entry))
      );
      setNotice(`Listing marked ${status.toLowerCase()}.`);
    } catch (e) {
      setError(e.message);
    }
  }

  async function resolveRequest(item) {
    if (!window.confirm(`Mark the ${item.animalType} assistance case as resolved?`)) return;
    setBusy(true);
    setError('');
    setNotice('');
    try {
      await api.updateReportStatus(token, item._id, 'Resolved');
      setNotice('Assistance request marked resolved.');
      await refresh();
    } catch (e) {
      setError(e.message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <DashboardLayout eyebrow="SHELTER DESK">
      <div className="stat-grid">
        <div className="stat-card">
          <span>Adoption listings</span>
          <strong>{listings.length}</strong>
        </div>
        <div className="stat-card">
          <span>Available</span>
          <strong>{listings.filter((item) => item.adoptionStatus === 'Available').length}</strong>
        </div>
        <div className="stat-card">
          <span>Incoming requests</span>
          <strong>{requests.length}</strong>
        </div>
        <div className="stat-card">
          <span>Profile status</span>
          <strong style={{ fontSize: 16, marginTop: 12 }}>{shelter ? 'Listed' : 'Set up'}</strong>
        </div>
      </div>

      {error && <Alert kind="error">{error}</Alert>}
      {notice && <Alert kind="success">{notice}</Alert>}

      {loading ? (
        <Loading label="Loading shelter workspaceâ€¦" />
      ) : (
        <>
          <PageSection
            title="Shelter information"
            detail="Keep contact details current for adopters and rescuers"
          >
            <form onSubmit={saveShelter} className="form-grid">
              <input type="hidden" name="location" value={JSON.stringify(shelterLocation)} />
              <div className="form-field">
                <label>Shelter name *</label>
                <input
                  name="name"
                  required
                  defaultValue={shelter?.name || ''}
                  placeholder="Your rescue center"
                />
              </div>
              <div className="form-field">
                <label>Contact phone *</label>
                <input name="phone" required defaultValue={shelter?.phone || user.phone || ''} />
              </div>
              <div className="form-field">
                <label>Email *</label>
                <input
                  name="email"
                  type="email"
                  required
                  defaultValue={shelter?.email || user.email || ''}
                />
              </div>
              <div className="form-field">
                <label>Address *</label>
                <input name="address" required defaultValue={shelter?.address || ''} />
              </div>
              <div className="form-field full">
                <label>Shelter location pin</label>
                <AnimalMap
                  location={shelterLocation}
                  onSelect={setShelterLocation}
                  markerLabel="Shelter location"
                />
                <span className="form-hint">
                  Choose the shelter's location on the map so people can find the right rescue center.
                </span>
              </div>
              <div className="form-field full">
                <label>Description *</label>
                <textarea
                  name="description"
                  required
                  defaultValue={shelter?.description || ''}
                  placeholder="What kind of care does your shelter provide?"
                />
              </div>
              <div className="form-field full">
                <button className="button button-primary" disabled={busy}>
                  {shelter ? 'Save shelter details' : 'Add shelter information'}
                </button>
              </div>
            </form>
          </PageSection>

          <PageSection
            title="Add an animal for adoption"
            detail="New listings are visible to the public while Available"
          >
            <form onSubmit={addListing} className="form-grid">
              <div className="form-field">
                <label>Animal type *</label>
                <input name="animalType" required placeholder="Indie dog, kittenâ€¦" />
              </div>
              <div className="form-field">
                <label>Age *</label>
                <input name="age" required placeholder="About 1 year" />
              </div>
              <div className="form-field">
                <label>Gender</label>
                <select name="gender" defaultValue="Unknown">
                  <option>Female</option>
                  <option>Male</option>
                  <option>Unknown</option>
                </select>
              </div>
              <div className="form-field">
                <label>Photo *</label>
                <input
                  type="file"
                  name="photo"
                  required
                  accept="image/jpeg,image/png,image/webp,image/gif"
                />
              </div>
              <div className="form-field full">
                <label>About this animal *</label>
                <textarea
                  name="description"
                  required
                  maxLength="1400"
                  placeholder="Personality, care needs, and what kind of home would suit them."
                />
              </div>
              <div className="form-field full">
                <button className="button button-primary" disabled={busy || !shelter}>
                  {busy ? 'Savingâ€¦' : 'Add adoption listing'}
                </button>
                {!shelter && (
                  <span className="form-hint">
                    Add your shelter profile before creating an adoption listing.
                  </span>
                )}
              </div>
            </form>
          </PageSection>

          <PageSection
            title="Adoption listings"
            detail="Update adoption status when an animal finds a home"
          >
            {listings.length ? (
              <div className="data-list">
                {listings.map((item) => (
                  <div key={item._id} className="data-row">
                    <div className="data-main">
                      <strong>
                        {item.animalType} Â· {item.age}
                      </strong>
                      <span>{item.description}</span>
                    </div>
                    <div className="inline-actions">
                      <StatusBadge value={item.adoptionStatus} />
                      {item.adoptionStatus === 'Available' ? (
                        <button
                          className="button button-outline button-small"
                          onClick={() => changeStatus(item, 'Adopted')}
                        >
                          Mark adopted
                        </button>
                      ) : (
                        <button
                          className="button button-outline button-small"
                          onClick={() => changeStatus(item, 'Available')}
                        >
                          Reopen
                        </button>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            ) : (
              <EmptyState
                title="No adoption listings yet"
                text="Add an animal after saving shelter details."
              />
            )}
          </PageSection>

          <PageSection
            title="Incoming assistance requests"
            detail="Open verified cases for shelter support"
          >
            {requests.length ? (
              <div className="data-list">
                {requests.map((item) => (
                  <div className="data-row" key={item._id}>
                    <div className="data-main">
                      <strong>
                        {item.animalType} Â· {item.location?.address}
                      </strong>
                      <span>{item.assistanceRequired}</span>
                      <span>
                        <PriorityBadge value={item.priority} />
                      </span>
                    </div>
                    <div className="inline-actions">
                      <StatusBadge value={item.status} />
                      <button
                        className="button button-outline button-small"
                        disabled={busy}
                        onClick={() => resolveRequest(item)}
                      >
                        Mark resolved
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            ) : (
              <EmptyState
                title="No open requests to manage"
                text="Verified rescue cases will appear here for shelter teams."
              />
            )}
          </PageSection>
        </>
      )}
    </DashboardLayout>
  );
}

const priorityRank = { Emergency: 0, Urgent: 1, Normal: 2 };
const nextStatus = { Accepted: 'In Progress', 'In Progress': 'Resolved' };
const nearbyRadiusKm = 50;

function distanceKm(from, to) {
  if (!to || to.lat == null || to.lng == null) return Infinity;
  const radians = (degrees) => (degrees * Math.PI) / 180;
  const latitudeDelta = radians(Number(to.lat) - from.lat);
  const longitudeDelta = radians(Number(to.lng) - from.lng);
  const a =
    Math.sin(latitudeDelta / 2) ** 2 +
    Math.cos(radians(from.lat)) *
      Math.cos(radians(Number(to.lat))) *
      Math.sin(longitudeDelta / 2) ** 2;
  return 6371 * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

export default function VolunteerDashboard() {
  const { token } = useAuth();
  const [queue, setQueue] = useState([]);
  const [assignments, setAssignments] = useState([]);
  const [profile, setProfile] = useState(null);
  const [availability, setAvailability] = useState('available');
  const [contact, setContact] = useState('');
  const [area, setArea] = useState('');
  const [nearbyCenter, setNearbyCenter] = useState(null);
  const [locating, setLocating] = useState(false);
  const [nearbyError, setNearbyError] = useState('');
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [busyId, setBusyId] = useState('');
  const refresh = useCallback(async () => {
    setError('');
    try {
      const [queueData, summary] = await Promise.all([
        api.volunteerRequests(token),
        api.volunteerSummary(token),
      ]);
      const sorted = (queueData.reports || []).sort(
        (a, b) =>
          priorityRank[a.priority] - priorityRank[b.priority] || new Date(a.date) - new Date(b.date)
      );
      setQueue(sorted);
      setAssignments(summary.assignments || []);
      setProfile(summary.profile);
      setAvailability(summary.profile?.availability || 'available');
      setContact(summary.profile?.contact || '');
      setArea(summary.profile?.area || '');
    } catch (e) {
      setError(e.message);
    } finally {
      setLoading(false);
    }

    refresh();
  }, [refresh]);

    setBusyId(id);

    setNotice('');

        'Rescue accepted. The reporterâ€™s phone and email are now in your assignment details.'
      );

      setBusyId('');

    setNotice('');

      await api.updateReportStatus(token, item._id, status);
      setNotice(`Case status updated to ${status}.`);
      await refresh();

      setError(e.message);
    } finally {
      setBusyId('');
    }

        setLocating(false);
